/* 68 byte */
struct cpu_ctx_t {
    uint32_t cpsr;
    uint32_t r0;
    uint32_t r1;
    uint32_t r2;
    uint32_t r3;
    uint32_t r4;        // callee saved
    uint32_t r5;        // callee saved
    uint32_t r6;        // callee saved
    uint32_t r7;        // callee saved
    uint32_t r8;        // callee saved
    uint32_t r9;        // callee saved
    uint32_t r10;       // callee saved
    uint32_t r11;       // callee saved 
    uint32_t r12;
    uint32_t lr;        // callee saved
};

.globl switch_fcontext
.type switch_fcontext, %function
switch_fcontext:
    /* save — use sp directly, no helper register needed */
    stmfd sp!, {r0-r12, lr}    /* push r0-r12 and lr    */
    mrs   r2,  cpsr
    stmfd sp!, {r2}            /* push cpsr             */
    str   sp,  [r0]            /* curr->context = sp    */

    /* restore — sp is the frame pointer, never changes until last */
    mov   sp,  r1              /* sp = next->context    */
    ldmfd sp!, {r2}            /* pop cpsr into r2      */
    msr   cpsr_cxsf, r2        /* restore cpsr          */
    ldmfd sp!, {r0-r12, lr}    /* pop r0-r12 and lr     */
    bx    lr                   /* jump                  */

struct task_t *create_task(const char *name, void *stack_mem, size_t stk_sz, void (*func)(void *), void *arg)
{
    struct task_t *t = malloc(sizeof(struct task_t));

    t->name       = strdup(name);

    t->v_accum     = 0;
    t->cycles_used = 0;
    t->yield_count = 0;
    t->preempt_count = 0;

    t->priority      = 128;              /* default mid priority */
    t->vruntime      = 0;               /* cfs_enqueue will set this */

    t->start_time  = cp15_read_cntvct(); 
    t->cycle_start = cp15_read_pmccntr();

    t->entry_func = func;
    t->user_arg   = arg;

    t->stack_base = stack_mem;
    t->stack_size = stk_sz;


    uint8_t *stack_top = (uint8_t *)stack_mem + stk_sz;
    stack_top = (uint8_t *)((uintptr_t)stack_top & ~7UL);

    stack_top -= sizeof(struct cpu_ctx_t); // reserve 36 byte
    struct cpu_ctx_t *frame = (struct cpu_ctx_t *)stack_top;

    frame->cpsr = 0x13;                            /* SVC mode, IRQs enabled */

    frame->r4 = 0xdeadbeef;
    frame->r5 = 0xdeadbeef;
    frame->r6 = 0xdeadbeef;
    frame->r7 = 0xdeadbeef;
    frame->r8 = 0xdeadbeef;
    frame->r9 = 0xdeadbeef;
    frame->r10 = 0xdeadbeef;
    frame->r11 = 0xdeadbeef;
    frame->lr = (uint32_t)task_entry_trampoline;

    t->context = stack_top;

    append_task(t);
    return t;
}

struct task_t *create_task(const char *name, void *stack_mem, size_t stk_sz, void (*func)(void *), void *arg)
{
    struct task_t *t = malloc(sizeof(struct task_t));

    t->name       = strdup(name);

    t->v_accum     = 0;
    t->cycles_used = 0;
    t->yield_count = 0;
    t->preempt_count = 0;

    t->priority      = 128;              /* default mid priority */
    t->vruntime      = 0;               /* cfs_enqueue will set this */

    t->start_time  = cp15_read_cntvct(); 
    t->cycle_start = cp15_read_pmccntr();

    t->entry_func = func;
    t->user_arg   = arg;

    t->stack_base = stack_mem;
    t->stack_size = stk_sz;


    uint8_t *stack_top = (uint8_t *)stack_mem + stk_sz;

    stack_top = (uint8_t *)((uintptr_t)stack_top & ~7UL);
    stack_top -= sizeof(struct cpu_ctx_t);          /* 328 bytes */
    struct cpu_ctx_t *frame = (struct cpu_ctx_t *)stack_top;

    memset(frame, 0, sizeof(struct cpu_ctx_t));

    frame->cpsr  = 0x13;                            /* SVC mode, IRQs enabled */
    frame->lr    = (uint32_t)task_entry_trampoline; /* bx lr jumps here       */
    frame->fpexc = (1U << 30);                      /* VFP enabled            */

    /* debug markers */
    frame->r4  = 0xdeadbeef;
    frame->r5  = 0xdeadbeef;
    frame->r6  = 0xdeadbeef;
    frame->r7  = 0xdeadbeef;
    frame->r8  = 0xdeadbeef;
    frame->r9  = 0xdeadbeef;
    frame->r10 = 0xdeadbeef;
    frame->r11 = 0xdeadbeef;

    for (int i = 8; i <= 15; i++) {
        frame->d[i] = 0xdeadbeefcafeebabe;
    }
    t->context = stack_top;

    append_task(t);
    return t;
}
.globl switch_fcontext
.type switch_fcontext, %function
switch_fcontext:
    /* push core first */
    stmfd sp!, {r0-r12, lr}
    mrs   r2,  cpsr
    stmfd sp!, {r2}

    /* push vfp control */
    vmrs  r2,  fpscr
    stmfd sp!, {r2}
    vmrs  r2,  fpexc
    stmfd sp!, {r2}

    /* push vfp data — split into two groups */
    vstmdb sp!, {d16-d31}       /* push d16-d31 first */
    vstmdb sp!, {d0-d15}        /* push d0-d15 last   */

    str   sp,  [r0]             /* curr->context = sp */

    /* restore next task */
    mov   sp,  r1

    /* pop vfp data */
    vldmia sp!, {d0-d15}        /* pop d0-d15 first   */
    vldmia sp!, {d16-d31}       /* pop d16-d31        */

    /* pop vfp control */
    ldmfd  sp!, {r2}
    vmsr   fpexc, r2
    ldmfd  sp!, {r2}
    vmsr   fpscr, r2

    /* pop core */
    ldmfd  sp!, {r2}
    msr    cpsr_cxsf, r2
    ldmfd  sp!, {r0-r12, lr}
    bx     lr

struct cpu_ctx_t {
    uint64_t d[32];
    uint32_t fpexc;
    uint32_t fpscr;
    uint32_t cpsr;
    uint32_t r0;
    uint32_t r1;
    uint32_t r2;
    uint32_t r3;
    uint32_t r4;
    uint32_t r5;
    uint32_t r6;
    uint32_t r7;
    uint32_t r8;
    uint32_t r9;
    uint32_t r10;
    uint32_t r11;
    uint32_t r12;
    uint32_t lr;
};

_Static_assert(sizeof(struct cpu_ctx_t) == 328, "cpu_ctx_t size wrong");
